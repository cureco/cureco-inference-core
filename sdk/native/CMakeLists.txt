cmake_minimum_required(VERSION 3.24)
project(cureco_inference VERSION 0.2.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
set(ONNXRUNTIME_ROOT "" CACHE PATH "Target ONNX Runtime SDK (include/ and lib/)")
set(CURECO_JSON_INCLUDE_DIR "" CACHE PATH "Directory containing nlohmann/json.hpp")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" REQUIRED NO_DEFAULT_PATH)
find_library(ORT_LIBRARY NAMES onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" REQUIRED NO_DEFAULT_PATH)
find_path(JSON_INCLUDE nlohmann/json.hpp PATHS "${CURECO_JSON_INCLUDE_DIR}" REQUIRED)
add_library(cureco_inference SHARED src/inference.cpp)
target_compile_features(cureco_inference PUBLIC cxx_std_17)
target_compile_definitions(cureco_inference PRIVATE CURECO_BUILD)
target_include_directories(cureco_inference PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_include_directories(cureco_inference PRIVATE "${ORT_INCLUDE}" "${JSON_INCLUDE}")
target_link_libraries(cureco_inference PRIVATE "${ORT_LIBRARY}")
set_target_properties(cureco_inference PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES INSTALL_RPATH "$ORIGIN")
if(MSVC)
  target_compile_options(cureco_inference PRIVATE /W4 /utf-8)
else()
  target_compile_options(cureco_inference PRIVATE -Wall -Wextra -Wpedantic)
endif()
install(TARGETS cureco_inference EXPORT CurecoTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT CurecoTargets FILE CurecoTargets.cmake NAMESPACE Cureco:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Cureco)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/CurecoConfig.cmake" "include(\"\${CMAKE_CURRENT_LIST_DIR}/CurecoTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/CurecoConfig.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Cureco)
if(BUILD_TESTING)
  add_executable(cureco_abi_test tests/abi.c)
  target_link_libraries(cureco_abi_test PRIVATE cureco_inference)
  add_test(NAME cureco_abi COMMAND cureco_abi_test)
endif()
